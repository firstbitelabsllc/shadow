  ```swift
  /// Outcome of a single polling step against a vendor.
  public enum ReceiptScanProgress: Sendable {
    /// Vendor finished and returned the parsed receipt.
    case completed(ScannedReceipt)
    /// Vendor accepted the request but is still processing. Caller should
    /// `pollOnce(operationId:)` after a delay (vendor-specific; ~1-2s for Azure).
    case inProgress(operationId: String)
    /// Vendor rejected the request or terminal-failed mid-process. The error
    /// shape matches what `scan()` would throw — surfaced here as a return
    /// value because polling sequences distinguish "transient incomplete"
    /// from "definitive failure" naturally at this seam.
    case failed(ReceiptScanError)
  }

  public protocol ReceiptScanProvider: Sendable {
    var providerName: String { get }
    var providerVersion: String { get }
    func scan(imageData: Data) async throws -> ScannedReceipt
    /// Begin a scan. Returns `.completed` if the vendor's start-call is
    /// blocking-until-terminal AND the operation finished within the call;
    /// returns `.inProgress(operationId:)` if the vendor is async; returns
    /// `.failed` for definitive failures (auth, malformed image, vendor outage).
    func start(imageData: Data) async throws -> ReceiptScanProgress
    /// Poll a previously-started operation. Caller is responsible for sleeping
    /// between polls (vendor-recommended interval, e.g. 2s for Azure DI v4).
    func pollOnce(operationId: String) async throws -> ReceiptScanProgress
  }

  extension ReceiptScanProvider {
    /// Default `scan()` for adapters that prefer to compose `start` + `pollOnce`.
    /// `AzureDIv4Provider` overrides with its own impl that leverages
    /// `uploadReceiptV4`'s internal 10× poll for efficiency on the simple path.
    public func scan(imageData: Data) async throws -> ScannedReceipt {
      var progress = try await start(imageData: imageData)
      while case .inProgress(let operationId) = progress {
        try await Task.sleep(for: .seconds(2))
        progress = try await pollOnce(operationId: operationId)
      }
      switch progress {
      case .completed(let receipt): return receipt
      case .failed(let error): throw error
      case .inProgress: preconditionFailure("loop terminates on non-inProgress")
      }
    }
  }
  ```
